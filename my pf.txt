    <script>
        document.getElementById('current-date').innerText = new Date().toUTCString();

        function showSection(id) {
            // Hide all
            document.querySelectorAll('.scroll-section').forEach(section => {
                section.classList.remove('active');
            });
            // Show target
            const target = document.getElementById(id);
            target.classList.add('active');
            
            // Scroll to bottom
            const terminalBody = document.getElementById('terminal-content');
            terminalBody.scrollTop = terminalBody.scrollHeight;

            // Update Input text simulation
            document.getElementById('typed-text').innerText = `open ${id.toUpperCase()} --force`;
        }

        function resetTerminal() {
            showSection('welcome');
            document.getElementById('typed-text').innerText = 'clear';
            setTimeout(() => {
                document.getElementById('typed-text').innerText = '';
            }, 1000);
        }
    </script>
